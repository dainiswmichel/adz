#!/bin/bash
# /opt/adz-flagship/install-wordpress-org.sh
# This file is the image entrypoint. The image CMD is not used.
set -euo pipefail
if [ ! -f /var/www/html/wp-includes/version.php ]; then
  ver="$(php -r '
    $j = json_decode(file_get_contents("https://api.wordpress.org/core/version-check/1.7/"), true);
    echo $j["offers"][0]["current"];
  ')"
  curl -fsSL -o /tmp/wordpress.tar.gz "https://wordpress.org/wordpress-${ver}.tar.gz"
  curl -fsSL -o /tmp/wordpress.tar.gz.sha1 "https://wordpress.org/wordpress-${ver}.tar.gz.sha1"
  echo "$(awk "{print \$1}" /tmp/wordpress.tar.gz.sha1)  /tmp/wordpress.tar.gz" | sha1sum -c -
  tar -xzf /tmp/wordpress.tar.gz -C /var/www/html --strip-components=1
  rm -rf /usr/src/wordpress /tmp/wordpress.tar.gz /tmp/wordpress.tar.gz.sha1
  chown -R www-data:www-data /var/www/html
fi
exec docker-entrypoint.sh apache2-foreground

wordpress:php8.3-apache supplies PHP 8.3 and Apache. Replacing only command leaves the image entrypoint, and that entrypoint copies /usr/src/wordpress into an empty volume before any later script runs. The script above is the entrypoint, so the wordpress.org tarball is the core. It then execs docker-entrypoint.sh, which writes wp-config.php from the WORDPRESS_DB_* variables and does not copy core, because version.php is already present. An existing volume is left as installed. The CLI image runs as uid 33, the same user as Apache, so a later dashboard update can write wp-content/upgrade.
